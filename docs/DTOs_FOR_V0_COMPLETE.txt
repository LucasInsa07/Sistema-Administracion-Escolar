Propósito:

- Mantener este archivo privado: no incluir secrets ni application.properties.

Instrucciones para v0.dev:
- Por favor, use los bloques de DTO exactamente como están abajo para mapear a TypeScript.
- No modifique los nombres de campos sin preguntar.
- Mapeo recomendado Java->TypeScript:
  - long/Long/int/Integer -> number
  - String -> string
  - LocalDate/Date -> string (ISO)
  - List<X> -> X[]
  - Enum -> string (o union de literales si es apropiado)

import jakarta.validation.constraints.Email;
import jakarta.validation.constraints.NotBlank;

public class AlumnoRequestDTO {
    @NotBlank(message = "El nombre no puede estar vacío")
    public String nombre;

    @NotBlank(message = "El apellido no puede estar vacío")
    public String apellido;

    @NotBlank(message = "El email no puede estar vacío")
    @Email(message = "El email debe tener un formato válido")
    public String email;
}

package com.demo.sistema_gestion_escolar.dto;

import jakarta.validation.constraints.NotBlank;
import jakarta.validation.constraints.NotNull;

public class MateriaRequestDTO {
    
    @NotBlank(message = "El nombre es obligatorio")
    private String nombre;
    
    @NotNull(message = "El profesor es obligatorio")
    private Long profesorId;

    // Constructores
    public MateriaRequestDTO() {}

    public MateriaRequestDTO(String nombre, Long profesorId) {
        this.nombre = nombre;
        this.profesorId = profesorId;
    }

    // Getters y setters
    public String getNombre() {
        return nombre;
    }

    public void setNombre(String nombre) {
        this.nombre = nombre;
    }

    public Long getProfesorId() {
        return profesorId;
    }

    public void setProfesorId(Long profesorId) {
        this.profesorId = profesorId;
    }
}

<<< END DTO
